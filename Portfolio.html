        function renderProjects(filter) {

            const grid = document.getElementById('project-grid');

            grid.innerHTML = '';

            

            const filtered = filter === 'all' 

                ? portfolioData.projects 

                : portfolioData.projects.filter(p => {

                    if(filter === 'AI') return p.category === 'AI';

                    return p.category === filter;

                });


            filtered.forEach(p => {

                const card = document.createElement('div');

                card.className = 'bg-white p-6 rounded-xl border border-slate-200 shadow-sm card-hover flex flex-col h-full';

                

                // Determine icon based on category

                let iconClass = 'fa-folder';

                let colorClass = 'text-slate-500';

                if(p.category === 'AI') { iconClass = 'fa-brain'; colorClass = 'text-purple-600'; }

                if(p.category === 'Healthcare') { iconClass = 'fa-heartbeat'; colorClass = 'text-emerald-600'; }

                if(p.category === 'Finance') { iconClass = 'fa-chart-line'; colorClass = 'text-blue-600'; }


                card.innerHTML = `

                    <div class="flex justify-between items-start mb-4">

                        <div class="p-2 bg-slate-50 rounded-lg ${colorClass}">

                            <i class="fas ${iconClass} text-xl"></i>

                        </div>

                        <span class="text-xs font-bold text-slate-400">${p.date}</span>

                    </div>

                    <h3 class="text-lg font-bold text-slate-800 mb-2 leading-tight">${p.title}</h3>

                    <p class="text-sm text-slate-600 mb-4 flex-grow">${p.desc}</p>

                    

                    <div class="border-t border-slate-100 pt-4 mt-auto">

                        <div class="flex flex-wrap gap-2">

                            ${p.tech.map(t => `<span class="text-[10px] font-semibold bg-slate-50 text-slate-500 px-2 py-1 rounded border border-slate-100">${t}</span>`).join('')}

                        </div>

                    </div>

                `;

                grid.appendChild(card);

            });

        }
